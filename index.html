/** * ============================================================================ * PURCHASE ENTRY — WEB APP BACKEND (v3) * Author: Somnath Karmakar * ============================================================================ * * A standalone Google Apps Script web app that backs an external HTML form * (hosted on GitHub Pages). It stores procurement orders in a Google Sheet, * one row per order, and exposes a small JSON API. * * DATA MODEL — Sheet "Purchases", one row per order * ---------------------------------------------------------------------------- * A Timestamp real date-time (dd-MM-yyyy HH:mm:ss) * B Receiver person who raised the requirement * C Order ID auto serial: REQ-0001, REQ-0002, ... * D Items human readable, one per line: "Cement | 5 Kg" * E Remarks free text * F _ItemsJSON HIDDEN helper. Exact items array so Edit reloads cleanly. * * Each item is { name, qty, unit } where unit ∈ {Kg, Nos, Litre, g, }. * Text fields (receiver / item names / remarks) are capitalized server-side. * The whole sheet renders in the Jost font. * * HTTP API * ---------------------------------------------------------------------------- * GET ?action=list → { ok, orders:[...] } newest 10, with edit state * GET → { ok, msg } health check * POST { action:'create' } → { ok, order } assigns Order ID, saves * POST { action:'update' } → { ok, order } | { ok:false, error } * edits an order ONLY within the edit window * * BUSINESS RULES * ---------------------------------------------------------------------------- * • Order IDs are issued under a script lock so concurrent submits never * collide on the same serial number. * • An order may be edited only within EDIT_WINDOW_MS of its creation. * The window is enforced here on the server — the client cannot bypass it. * * DEPLOYMENT * ---------------------------------------------------------------------------- * Run any function once → Authorize → Deploy → New deployment → Web app → * Execute as: Me → Who has access: Anyone → Deploy → copy the /exec URL. * After ANY code change: Manage deployments → edit → Version: New version. * ============================================================================ */ /* ---------------------------------------------------------------------------- * CONFIGURATION * ------------------------------------------------------------------------- */ var CONFIG = { SHEET_NAME: 'Purchases', HEADERS: ['Timestamp', 'Receiver', 'Order ID', 'Items', 'Remarks', '_ItemsJSON'], EDIT_WINDOW_MS: 2 * 60 * 60 * 1000, // 2 hours ID_PREFIX: 'REQ-', ID_PAD: 4, FONT: 'Jost', TS_FORMAT: 'dd-MM-yyyy HH:mm:ss', DEFAULT_UNIT: 'Nos', LOCK_WAIT_MS: 5000 }; // Column positions (1-based). Single source of truth for the layout. var COL = { TS: 1, RECEIVER: 2, ORDER_ID: 3, ITEMS: 4, REMARKS: 5, JSON: 6 }; /* ---------------------------------------------------------------------------- * HTTP ENTRY POINTS * ------------------------------------------------------------------------- */ function doGet(e) { var action = (e && e.parameter && e.parameter.action) || ''; var sheet = SheetRepo.get(); if (action === 'list') return Api.json(OrderService.list(sheet)); return Api.json({ ok: true, msg: 'Purchase backend live' }); } function doPost(e) { try { var payload = JSON.parse(e.postData.contents); var sheet = SheetRepo.get(); switch (payload.action) { case 'update': return Api.json(OrderService.update(sheet, payload)); case 'list': return Api.json(OrderService.list(sheet)); case 'create': default: return Api.json(OrderService.create(sheet, payload)); } } catch (err) { return Api.json({ ok: false, error: String(err) }); } } /* ---------------------------------------------------------------------------- * API HELPER * ------------------------------------------------------------------------- */ var Api = { json: function (obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } }; /* ---------------------------------------------------------------------------- * SHEET REPOSITORY — owns the physical sheet & its formatting * ------------------------------------------------------------------------- */ var SheetRepo = { get: function () { var ss = SpreadsheetApp.getActive(); var sh = ss.getSheetByName(CONFIG.SHEET_NAME) || ss.insertSheet(CONFIG.SHEET_NAME); if (sh.getLastRow() === 0) this._initialise(sh); return sh; }, _initialise: function (sh) { sh.appendRow(CONFIG.HEADERS); sh.getRange(1, 1, 1, CONFIG.HEADERS.length) .setFontWeight('bold').setBackground('#1e3a8a').setFontColor('#ffffff'); sh.setFrozenRows(1); sh.getRange('A1:F2000').setFontFamily(CONFIG.FONT); sh.hideColumns(COL.JSON); sh.setColumnWidth(COL.ITEMS, 240); }, styleRow: function (sh, rowIndex) { sh.getRange(rowIndex, 1, 1, CONFIG.HEADERS.length) .setFontFamily(CONFIG.FONT).setVerticalAlignment('middle'); sh.getRange(rowIndex, COL.TS).setNumberFormat(CONFIG.TS_FORMAT); sh.getRange(rowIndex, COL.ITEMS).setWrap(true); }, /** Find the 1-based sheet row for an Order ID, or -1 if absent. */ findRowByOrderId: function (sh, orderId) { var last = sh.getLastRow(); if (last < 2) return -1; var ids = sh.getRange(2, COL.ORDER_ID, last - 1, 1).getValues(); for (var i = 0; i < ids.length; i++) { if (String(ids[i][0]) === String(orderId)) return i + 2; } return -1; }, /** Next serial Order ID, derived from the highest existing number. */ nextOrderId: function (sh) { var last = sh.getLastRow(); var max = 0; if (last >= 2) { var ids = sh.getRange(2, COL.ORDER_ID, last - 1, 1).getValues(); for (var i = 0; i < ids.length; i++) { var m = /(\d+)\s*$/.exec(String(ids[i][0] || '')); if (m) { var n = parseInt(m[1], 10); if (n > max) max = n; } } } return CONFIG.ID_PREFIX + String(max + 1).padStart(CONFIG.ID_PAD, '0'); } }; /* ---------------------------------------------------------------------------- * ORDER SERVICE — business logic (create / list / update) * ------------------------------------------------------------------------- */ var OrderService = { list: function (sh) { var last = sh.getLastRow(); var orders = []; if (last >= 2) { var rows = sh.getRange(2, 1, last - 1, CONFIG.HEADERS.length).getValues(); for (var i = 0; i < rows.length; i++) { if (!rows[i][COL.ORDER_ID - 1]) continue; orders.push(OrderMapper.fromRow(rows[i])); } } orders.sort(function (a, b) { return new Date(b.createdAt) - new Date(a.createdAt); }); return { ok: true, orders: orders.slice(0, 10) }; }, create: function (sh, payload) { var lock = LockService.getScriptLock(); lock.waitLock(CONFIG.LOCK_WAIT_MS); try { var order = { orderId: SheetRepo.nextOrderId(sh), createdAt: new Date(), receiver: Text.capitalize(payload.receiver), remarks: Text.capitalize(payload.remarks), items: Items.clean(payload.items) }; sh.appendRow([ order.createdAt, order.receiver, order.orderId, Items.toReadable(order.items), order.remarks, JSON.stringify(order.items) ]); SheetRepo.styleRow(sh, sh.getLastRow()); return { ok: true, order: OrderMapper.toClient(order, order.createdAt) }; } finally { lock.releaseLock(); } }, update: function (sh, payload) { if (!payload.orderId) return { ok: false, error: 'No orderId supplied.' }; var rowIndex = SheetRepo.findRowByOrderId(sh, payload.orderId); if (rowIndex < 0) return { ok: false, error: 'Order ' + payload.orderId + ' was not found.' }; // Enforce the edit window, anchored to the (immutable) creation timestamp. var createdAt = sh.getRange(rowIndex, COL.TS).getValue(); var age = Date.now() - new Date(createdAt).getTime(); if (isNaN(age) || age > CONFIG.EDIT_WINDOW_MS) { return { ok: false, error: 'Editing window closed. The 2-hour limit has passed and this order can no longer be edited.' }; } var items = Items.clean(payload.items); var receiver = Text.capitalize(payload.receiver); var remarks = Text.capitalize(payload.remarks); // Timestamp and Order ID are immutable; only editable fields change. sh.getRange(rowIndex, COL.RECEIVER).setValue(receiver); sh.getRange(rowIndex, COL.ITEMS).setValue(Items.toReadable(items)); sh.getRange(rowIndex, COL.REMARKS).setValue(remarks); sh.getRange(rowIndex, COL.JSON).setValue(JSON.stringify(items)); SheetRepo.styleRow(sh, rowIndex); var row = sh.getRange(rowIndex, 1, 1, CONFIG.HEADERS.length).getValues()[0]; return { ok: true, order: OrderMapper.fromRow(row) }; } }; /* ---------------------------------------------------------------------------- * MAPPERS — translate between sheet rows and client-facing objects * ------------------------------------------------------------------------- */ var OrderMapper = { // Sheet row → client object (used by list & update responses) fromRow: function (r) { var createdIso = Time.toIso(r[COL.TS - 1]); var items = []; try { items = JSON.parse(r[COL.JSON - 1] || '[]'); } catch (e) { items = []; } return this._withEditState({ orderId: r[COL.ORDER_ID - 1], receiver: r[COL.RECEIVER - 1], items: items, remarks: r[COL.REMARKS - 1], createdAt: createdIso }, createdIso); }, // In-memory order → client object (used by create response) toClient: function (order, createdAtDate) { var createdIso = Time.toIso(createdAtDate); return this._withEditState({ orderId: order.orderId, receiver: order.receiver, items: order.items, remarks: order.remarks, createdAt: createdIso }, createdIso); }, _withEditState: function (obj, createdIso) { var msLeft = CONFIG.EDIT_WINDOW_MS - (Date.now() - new Date(createdIso).getTime()); if (isNaN(msLeft)) msLeft = 0; obj.editable = msLeft > 0; obj.msLeft = Math.max(0, msLeft); return obj; } }; /* ---------------------------------------------------------------------------- * VALUE HELPERS * ------------------------------------------------------------------------- */ var Text = { // Capitalize the first letter of each word, leaving the rest as typed. capitalize: function (s) { return String(s == null ? '' : s).trim() .replace(/\b\w/g, function (c) { return c.toUpperCase(); }); } }; var Items = { // Normalize the raw items array from the client into clean {name, qty, unit}. clean: function (rawItems) { return (rawItems || []) .map(function (i) { return { name: Text.capitalize((i.name || '').trim()), qty: String(i.qty == null ? '' : i.qty).trim(), unit: String(i.unit == null ? '' : i.unit).trim() || CONFIG.DEFAULT_UNIT }; }) .filter(function (i) { return i.name !== ''; }); }, // "Cement | 5 Kg" per line. toReadable: function (items) { return items.map(function (i) { return i.name + ' | ' + i.qty + ' ' + i.unit; }).join('\n'); } }; var Time = { toIso: function (v) { if (v instanceof Date) return v.toISOString(); return String(v || ''); } };
